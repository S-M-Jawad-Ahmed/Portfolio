Feature Engineering 
-- Replace 'your_project_id' with your actual BigQuery project ID
              
CREATE OR REPLACE TABLE `your_project_id.divvy_dataset.all_trips_featured` AS
SELECT
  *,
  ROUND(TIMESTAMP_DIFF(TIMESTAMP(ended_at), TIMESTAMP(started_at), SECOND) / 60.0, 2) AS trip_duration_minutes,
  CASE 
    WHEN ROUND(TIMESTAMP_DIFF(TIMESTAMP(ended_at), TIMESTAMP(started_at), SECOND) / 60.0, 2) < 5 THEN 'Very Short'
    WHEN ROUND(TIMESTAMP_DIFF(TIMESTAMP(ended_at), TIMESTAMP(started_at), SECOND) / 60.0, 2) BETWEEN 5 AND 20 THEN 'Short'
    WHEN ROUND(TIMESTAMP_DIFF(TIMESTAMP(ended_at), TIMESTAMP(started_at), SECOND) / 60.0, 2) BETWEEN 20 AND 60 THEN 'Medium'
    ELSE 'Long'
  END AS trip_duration_category,
  FORMAT_DATE('%A', DATE(started_at)) AS trip_day_name,
  FORMAT_DATE('%B', DATE(started_at)) AS trip_month_name,
  EXTRACT(YEAR FROM started_at) AS trip_year,
  EXTRACT(HOUR FROM started_at) AS started_at_hour,
  CASE 
    WHEN EXTRACT(DAYOFWEEK FROM started_at) IN (1, 7) THEN 'Weekend'
    ELSE 'Weekday'
  END AS day_type,
  CASE 
    WHEN EXTRACT(MONTH FROM started_at) IN (12, 1, 2) THEN 'Winter'
    WHEN EXTRACT(MONTH FROM started_at) IN (3, 4, 5) THEN 'Spring'
    WHEN EXTRACT(MONTH FROM started_at) IN (6, 7, 8) THEN 'Summer'
    ELSE 'Fall'
  END AS season
FROM `your_project_id.divvy_dataset.all_trips_optimized`;

-- Figure 2_Percentage of Rider distribution_SQL
-- Replace 'your_project_id' with your actual BigQuery project ID

WITH rider_distribution AS (
  SELECT
    SUM(CASE WHEN member_casual = 'member' THEN 1 ELSE 0 END) AS member_trips,
    SUM(CASE WHEN member_casual = 'casual' THEN 1 ELSE 0 END) AS casual_trips,
    COUNT(*) AS total_trips
  FROM `your_project_id.divvy_dataset.divvy_trips_cleaned`
)

SELECT
  member_trips,
  ROUND(member_trips * 100.0 / total_trips, 2) AS percentage_member_trips,
  casual_trips,
  ROUND(casual_trips * 100.0 / total_trips, 2) AS percentage_casual_trips
FROM rider_distribution;

# Figure 6_Trip Duration Bucket_R

#Create duration_bucket
divvy_data <- divvy_data %>%
  mutate(duration_bucket = case_when(
    trip_duration_minutes < 5 ~ "<5 min",
    trip_duration_minutes >= 5 & trip_duration_minutes < 10 ~ "5-10 min",
    trip_duration_minutes >= 10 & trip_duration_minutes < 20 ~ "10-20 min",
    trip_duration_minutes >= 20 & trip_duration_minutes < 30 ~ "20-30 min",
    trip_duration_minutes >= 30 & trip_duration_minutes <= 60 ~ "30-60 min",
    trip_duration_minutes > 60 ~ ">60 min"
  ))

#Convert in order
divvy_data$duration_bucket <- factor(
  divvy_data$duration_bucket,
  levels = c("<5 min", "5-10 min", "10-20 min", "20-30 min", "30-60 min", ">60 min")
)

#Figure 8_Weekend vs Weekday Trips Trend_R

#Create hour_day
hour_day <- divvy_data %>%
  mutate(
    started_at_hour = hour(started_at),
    day_type = if_else(weekdays(started_at) %in% c("Saturday", "Sunday"), "Weekend", "Weekday")
  ) %>%
  group_by(day_type, started_at_hour) %>%
  summarise(trips_count = n(), .groups = "drop")

#Create heatmap
heatmap <- ggplot(hour_day, aes(x=started_at_hour, y=day_type, fill=trips_count)) +
    geom_tile(color="white", size=0.1) +                    # small border
    geom_text(aes(label=comma(trips_count)), size=3, angle=90, hjust=0.5, vjust=0.5) +  # vertical labels
    scale_fill_gradient(low="#FFF7BC", high="#D95F0E", labels=comma) +  # gradient
    labs(title="Heatmap of Trips by Hour and Day Type",
         x="Hour", y="Day Type", fill="Trips Count") +
    scale_x_continuous(breaks=0:23, expand=c(0,0)) +
    scale_y_discrete(expand=c(0,0)) +
    theme_minimal(base_size=12) +
    theme(axis.text=element_text(size=10),
          axis.title=element_text(size=12),
          plot.title=element_text(size=14, face="bold"))

print(heatmap)
